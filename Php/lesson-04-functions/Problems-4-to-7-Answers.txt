<?php

function getSum($a,$b,$c){
    return $a+$b+$c;
}

function getDifference($a,$b,$c){
    return $a-$b-$c;
}

function getProduct($a,$b,$c){
    return $a*$b*$c;
}

function getDivision($a,$b,$c){
    if($b==0 || $c==0){
        return "Cannot divide by zero";
    }
    return $a/$b/$c;
}

function getMaximum($a,$b,$c){
    $Largest = $a;
    if($b > $Largest){
        $Largest = $b;
    }
    if($c > $Largest){
        $Largest = $c;
    }
    return $Largest;
}

function getMinimum($a,$b,$c){
    $Smallest = $a;
    if($b < $Smallest){
        $Smallest = $b;
    }
    if($c < $Smallest){
        $Smallest = $c;
    }
    return $Smallest;
}

function getAverage($a,$b,$c){
    return ($a+$b+$c)/3;
}

?>

<?php

function isPalindrome($text){

    $clean = "";
    for($i=0; $i<strlen($text); $i++){
        $ch = $text[$i];
        if($ch != " "){
            $clean = $clean.strtoupper($ch);
        }
    }

    $reverse = "";
    for($i=strlen($clean)-1; $i>=0; $i--){
        $reverse = $reverse.$clean[$i];
    }

    if($clean == $reverse){
        return true;
    }
    return false;
}
?>

function checkPassword($password){

    $long = false;
    $upper = false;
    $lower = false;
    $digit = false;

    if(strlen($password) >= 8){
        $long = true;
    }

    for($i=0; $i<strlen($password); $i++){
        $ch = $password[$i];
        if($ch >= "A" && $ch <= "Z"){
            $upper = true;
        }
        if($ch >= "a" && $ch <= "z"){
            $lower = true;
        }
        if($ch >= "0" && $ch <= "9"){
            $digit = true;
        }
    }

    if($long && $upper && $lower && $digit){
        return "your password is valid";
    }
    return "Your password is not valid, please try again!";
}
?>

<?php

function isPalindromeNumber($n){

    $reverse = 0;
    $temp = $n;

    while($temp > 0){
        $digit = $temp % 10;
        $reverse = ($reverse * 10) + $digit;
        $temp = (int)($temp / 10);
    }

    if($reverse == $n){
        return true;
    }
    return false;
}

function isEven($n){
    if($n % 2 == 0){
        return true;
    }
    return false;
}

function isArmstrong($n){

    $sum = 0;
    $temp = $n;

    while($temp > 0){
        $digit = $temp % 10;
        $sum = $sum + ($digit * $digit * $digit);
        $temp = (int)($temp / 10);
    }

    if($sum == $n){
        return true;
    }
    return false;
}

function isPrime($n){

    if($n < 2){
        return false;
    }

    for($i=2; $i<=$n/2; $i++){
        if($n % $i == 0){
            return false;
        }
    }
    return true;
}
?>
